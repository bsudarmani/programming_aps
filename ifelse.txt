## **C & C++ Questions**
### **Basic If Condition Traps**
1. **Assignment inside if condition**
   ```c
   #include <stdio.h>
   int main() {
       int x = 3;
       if (x = 5) // Assignment instead of comparison
           printf("x is 5");
       else
           printf("x is not 5");
       return 0;
   }
   ```

2. **Logical Operators Confusion**
   ```c
   #include <stdio.h>
   int main() {
       int a = 0;
       if (a = 2 && a = 3) // Assignments take place, not comparisons
           printf("Condition Met");
       else
           printf("Condition Not Met");
       return 0;
   }
   ```

3. **Order of Operations in Condition**
   ```c
   #include <stdio.h>
   int main() {
       int a = 1, b = 2, c = 3;
       if (a < b < c) // Treated as (a < b) < c
           printf("Yes");
       else
           printf("No");
       return 0;
   }
   ```

4. **Bitwise Operators Confusion**
   ```c
   #include <stdio.h>
   int main() {
       int x = 1;
       if (x & 1) // Bitwise AND operation
           printf("Odd");
       else
           printf("Even");
       return 0;
   }
   ```

5. **Negative Condition Check**
   ```c
   #include <stdio.h>
   int main() {
       int x = -1;
       if (~x == 0) // Bitwise NOT operation
           printf("True");
       else
           printf("False");
       return 0;
   }
   ```

### **Loops with If Conditions**
6. ```c
   #include <stdio.h>
   int main() {
       for (int i = 0; i < 5; i++)
           if (i % 2 == 0)
               printf("%d ", i);
       return 0;
   }
   ```

7. ```c
   #include <stdio.h>
   int main() {
       int i = 0;
       while (i < 5)
           if (i++ == 3)
               printf("Three ");
           else
               printf("Loop ");
       return 0;
   }
   ```

### **Pointer Tricks**
8. ```c
   #include <stdio.h>
   int main() {
       int a = 10, *p;
       p = &a;
       if (*p == 10)
           printf("Pointer works");
       else
           printf("Pointer fails");
       return 0;
   }
   ```

9. ```c
   #include <stdio.h>
   int main() {
       char *p = "Hello";
       if (*p == 'H')
           printf("First letter is H");
       return 0;
   }
   ```

    }
    ```
