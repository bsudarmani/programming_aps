## **C & C++ Tricky Ternary Operator Questions**
### **Basic Traps**
1. **Simple Ternary Expression**
   ```c
   #include <stdio.h>
   int main() {
       int x = 5;
       int y = (x > 3) ? 10 : 20;
       printf("%d\n", y);
       return 0;
   }
   ```
   *(Hint: If `x` is greater than `3`, `y = 10`; otherwise, `y = 20`.)*

2. **Ternary with Pre/Post-Increment**
   ```c
   #include <stdio.h>
   int main() {
       int x = 3;
       int y = (x++ > 3) ? ++x : --x;
       printf("%d %d\n", x, y);
       return 0;
   }
   ```
   *(Hint: What is `x++` doing before evaluation?)*

3. **Nested Ternary Operator**
   ```c
   #include <stdio.h>
   int main() {
       int x = 4;
       int y = (x > 5) ? 100 : ((x < 3) ? 50 : 75);
       printf("%d\n", y);
       return 0;
   }
   ```
   *(Hint: This is equivalent to an if-else-if ladder.)*

### **Mixing Operators**
4. **Ternary with Bitwise AND**
   ```c
   #include <stdio.h>
   int main() {
       int a = 5, b = 3;
       int c = (a & b) ? 10 : 20;
       printf("%d\n", c);
       return 0;
   }
   ```
   *(Hint: What is `5 & 3` in binary?)*

5. **Ternary with Logical NOT**
   ```c

       int x = 0;
       int y = (!x) ? 1 : 2;
       printf("%d\n", y);
       return 0;
   }
   ```

6. **Ternary with Bitwise XOR**

   ```c
   }
   ```

### **Ternary with Loops**
7. **Using Ternary in `for` Loop**
   ```c
   #include <stdio.h>
   int main() {
       for (int i = 0; i < 5; i++)
           printf("%s ", (i % 2 == 0) ? "Even" : "Odd");
       return 0;
   }
   ```

8. **Ternary in While Loop Condition**
   ```c
   #include <stdio.h>
   int main() {
       int i = 3;
       while ((i-- > 0) ? 1 : 0)
           printf("%d ", i);
       return 0;
   }
   ```

9. **Ternary with Do-While Loop**
   ```c
   #include <stdio.h>
   int main() {
       int x = 2;
       do {
           printf("%d ", x);
           x = (x > 1) ? x - 1 : x + 1;
       } while (x > 0);
       return 0;
   }
   ```


