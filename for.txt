## **C & C++ Tricky For Loop Questions**
### **Basic Loop Traps**
//double initialiation,increment is possible 
//double condition is possible
//i=i++,return i++ 
1. **Loop with No Condition**
   ```c
   #include <stdio.h>
   int main() {
       for (;;)
           printf("Hello");
   }
   ```
   **(What happens?)**

2. **Unusual Increment**
   ```c
   #include <stdio.h>
   int main() {
       for (int i = 0; i < 5; i = i++) {
           printf("%d ", i);
       }
       return 0;
   }
   ```

3. **Multiple Updates**
   ```c
   #include <stdio.h>
   int main() {
       for (int i = 0, j = 5; i < j; i++, j--) {
           printf("%d %d\n", i, j);
       }
       return 0;
   }
   ```

4. **Comma Operator**
   ```c
   #include <stdio.h>
   int main() {
       for (int i = 0, j = 10; i < 5, j > 5; i++, j--) {
           printf("%d %d\n", i, j);
       }
       return 0;
   }
   ```

5. **Loop Skipping Values**
   ```c
   #include <stdio.h>
   int main() {
       for (int i = 1; i < 10; i *= 2) {
           printf("%d ", i);
       }
       return 0;
   }
   ```

### **Loop with Break and Continue**
6. **Break Inside For**
   ```c
   #include <stdio.h>
   int main() {
       for (int i = 0; i < 5; i++) {
           printf("%d ", i);
           if (i == 2) break;
       }
       return 0;
   }
   ```

7. **Continue with Nested Loops**
   ```c
   #include <stdio.h>
   int main() {
       for (int i = 0; i < 3; i++) {
           for (int j = 0; j < 3; j++) {
               if (j == 1) continue;
               printf("%d %d\n", i, j);
           }
       }
       return 0;
   }
   ```

---

### **For Loop with Function Calls**
8. ```c
   #include <stdio.h>
   int fun() {
       static int x = 0;
       return x++;
   }
   int main() {
       for (int i = 0; fun() < 3; i++)
           printf("%d ", i);
       return 0;
   }
   ```

9. **Recursion inside for-loop**
   ```c
   #include <stdio.h>
   void fun(int x) {
       if (x < 3) {
           printf("%d ", x);
           fun(x + 1);
       }
   }
   int main() {
       for (int i = 0; i < 2; i++)
           fun(i);
       return 0;
   }
   ```

### **Loop with Conditions and Operators**
10. **Using Assignment in Condition**
    ```c
    #include <stdio.h>
    int main() {
        int x = 0;
        for (; x = 5; ) {
            printf("Hello ");
            break;
        }
        return 0;
    }
    ```

11. **Nested Loops with Increment**
    ```c
    #include <stdio.h>
    int main() {
        for (int i = 0; i < 3; i++)
            for (int j = 0; j < 3; j++)
                printf("%d ", i + j);
        return 0;
    }
    ```

12. **Loop Running Backward**
    ```c
    #include <stdio.h>
    int main() {
        for (int i = 10; i >= 0; i -= 2) {
            printf("%d ", i);
        }
        return 0;
    }
    ```

## **C++ Tricky For Loop Questions**
### **Loop with STL Containers**
13. **Loop with Vector**
    ```cpp
    #include <iostream>
    #include <vector>
    using namespace std;
    int main() {
        vector<int> v = {10, 20, 30};
        for (int x : v)
            cout << x << " ";
        return 0;
    }
    ```

14. **Auto in Range-based Loop**
    ```cpp
    #include <iostream>
    using namespace std;
    int main() {
        int arr[] = {1, 2, 3, 4, 5};
        for (auto x : arr)
            cout << x << " ";
        return 0;
    }
    ```

15. **Loop with Pointers**
    ```cpp
    #include <iostream>
    using namespace std;
    int main() {
        char str[] = "Hello";
        for (char *p = str; *p; p++)
            cout << *p;
        return 0;
    }
    ```

## **Java Tricky For Loop Questions**
### **For Loop with Different Iteration Style**
16. **Infinite Loop**
    ```java
    public class Main {
        public static void main(String[] args) {
            for (;;)
                System.out.println("Hello");
        }
    }
    ```

17. **For Loop with String**
    ```java
    public class Main {
        public static void main(String[] args) {
            String str = "Java";
            for (int i = 0; i < str.length(); i++)
                System.out.print(str.charAt(i) + " ");
        }
    }
    ```

18. **Multiple Variables in For Loop**
    ```java
    public class Main {
        public static void main(String[] args) {
            for (int i = 0, j = 5; i < j; i++, j--)
                System.out.println(i + " " + j);
        }
    }
    ```

19. **Enhanced For Loop**
    ```java
    public class Main {
        public static void main(String[] args) {
            int arr[] = {10, 20, 30};
            for (int x : arr)
                System.out.print(x + " ");
        }
    }
    ```

20. **For Loop Inside a While**
    ```java
    public class Main {
        public static void main(String[] args) {
            int i = 0;
            while (i < 2) {
                for (int j = 0; j < 3; j++)
                    System.out.print(i + "" + j + " ");
                i++;
            }
        }
    }
    ```
