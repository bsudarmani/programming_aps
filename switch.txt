## **C & C++ Questions**
### **Basic Switch-Case Traps**
1. **No `break` statement**
   ```c
   #include <stdio.h>
   int main() {
       int x = 2;
       switch (x) {
           case 1: printf("One ");
           case 2: printf("Two ");
           case 3: printf("Three ");
       }
       return 0;
   }
   ```
   **Output?**

2. **Default Positioning**
   ```c
   #include <stdio.h>
   int main() {
       int x = 3;
       switch (x) {
           case 1: printf("One ");
           default: printf("Default ");
           case 3: printf("Three ");
       }
       return 0;
   }
   ```

3. **Character as a Case**
   ```c
   #include <stdio.h>
   int main() {
       char ch = 'B';
       switch (ch) {
           case 'A': printf("A ");
           case 'B': printf("B ");
           case 'C': printf("C ");
       }
       return 0;
   }
   ```

4. **Expression in Case**
   ```c
   #include <stdio.h>
   int main() {
       int x = 2;
       switch (x + 1) {
           case 1: printf("One ");
           case 3: printf("Three ");
           default: printf("Default ");
       }
       return 0;
   }
   ```

5. **Break Inside a Condition**
   ```c
   #include <stdio.h>
   int main() {
       int x = 2;
       switch (x) {
           case 1: printf("One ");
                   if (x == 2) break;
           case 2: printf("Two ");
           case 3: printf("Three ");
       }
       return 0;
   }
   ```

### **Switch with Loops**
6. **Using `switch` inside a loop**
   ```c
   #include <stdio.h>
   int main() {
       for (int i = 0; i < 3; i++) {
           switch (i) {
               case 0: printf("Zero "); break;
               case 1: printf("One "); break;
               default: printf("Other ");
           }
       }
       return 0;
   }
   ```

7. **Using Switch Inside While**
   ```c
   #include <stdio.h>
   int main() {
       int i = 0;
       while (i < 3) {
           switch (i) {
               case 0: printf("Zero ");
               case 1: printf("One ");
           }
           i++;
       }
       return 0;
   }
   ```

## **C++ Questions**
### **Switch with Variables and Scopes**
9. ```cpp
   #include <iostream>
   using namespace std;
   int main() {
       int x = 1;
       switch (x) {
           int y = 10;
           case 1: cout << y; break;
           case 2: cout << "Two"; break;
       }
       return 0;
   }
   ```

10. **Duplicate Cases**
    ```cpp
    #include <iostream>
    using namespace std;
    int main() {
        int x = 3;
        switch (x) {
            case 3: cout << "Three"; break;
            case 3: cout << "Duplicate Three"; break;
            default: cout << "Default";
        }
        return 0;
    }
    ```

## **Java Questions**
### **Integer Cases**
11. ```java
    public class Main {
        public static void main(String[] args) {
            int x = 2;
            switch (x) {
                case 1: System.out.print("One ");
                case 2: System.out.print("Two ");
                case 3: System.out.print("Three ");
            }
        }
    }
    ```

12. **Floating Point in Switch (Invalid)**
    ```java
    public class Main {
        public static void main(String[] args) {
            float x = 2.5f;
            switch (x) { // Compilation error
                case 2.5f: System.out.print("Two Point Five"); break;
            }
        }
    }
    ```

13. **Switch with Strings**
    ```java
    public class Main {
        public static void main(String[] args) {
            String day = "Monday";
            switch (day) {
                case "Sunday": System.out.println("Weekend"); break;
                case "Monday": System.out.println("Weekday"); break;
                default: System.out.println("Invalid");
            }
        }
    }
    ```

14. **Switch with Boolean (Invalid)**
    ```java
    public class Main {
        public static void main(String[] args) {
            boolean flag = true;
            switch (flag) { // Compilation error
                case true: System.out.println("Yes"); break;
                case false: System.out.println("No"); break;
            }
        }
    }
    ```

15. **Using `default` first**
    ```java
    public class Main {
        public static void main(String[] args) {
            int x = 3;
            switch (x) {
                default: System.out.println("Default ");
                case 3: System.out.println("Three ");
            }
        }
    }
    ```

### **Advanced Java Cases**
16. **Switch with Enum**
    ```java
    public class Main {
        enum Color { RED, GREEN, BLUE }
        public static void main(String[] args) {
            Color c = Color.GREEN;
            switch (c) {
                case RED: System.out.println("Red"); break;
                case GREEN: System.out.println("Green"); break;
                case BLUE: System.out.println("Blue"); break;
            }
        }
    }
    ```

17. **Switch inside a loop**
    ```java
    public class Main {
        public static void main(String[] args) {
            for (int i = 0; i < 3; i++) {
                switch (i) {
                    case 0: System.out.print("Zero "); break;
                    case 1: System.out.print("One "); break;
                    default: System.out.print("Other ");
                }
            }
        }
    }
    ```

18. **Switch with method calls**
    ```java
    public class Main {
        public static void main(String[] args) {
            int x = getNumber();
            switch (x) {
                case 1: System.out.println("One"); break;
                case 2: System.out.println("Two"); break;
                default: System.out.println("Other");
            }
        }
        static int getNumber() {
            return 2;
        }
    }
    ```
