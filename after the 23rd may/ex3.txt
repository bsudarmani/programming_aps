### ✅ **Q1. Fibonacci Explosion**

```java
public class Q1 {
    static int fib(int n) {
        if (n <= 1) return n;
```

### ✅ **Q15. Recursive Print All Palindromic Partitions**

```java
public class Q15 {
    static boolean isPalindrome(String s) {
        return s.equals(new StringBuilder(s).reverse().toString());
    }
    static void partition(String str, String res) {
        if (str.isEmpty()) {
            System.out.println(res);
            return;
        }
        for (int i = 1; i <= str.length(); i++) {
            String prefix = str.substring(0, i);
            if (isPalindrome(prefix)) {
                partition(str.substring(i), res + " (" + prefix + ")");
            }
        }
    }
    public static void main(String[] args) {
        partition("aabbaa", ""); // 40+ calls due to branching
    }
}
```
