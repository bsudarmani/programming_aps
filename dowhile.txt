## **C & C++ Tricky Do-While Loop Questions**
### **Basic Traps**
1. **Simple Do-While**
   ```c
   #include <stdio.h>
   int main() {
       int x = 1;
       do {
           printf("%d ", x);
       } while (x > 5);
       return 0;
   }
   ```
   **(Hint: Executes at least once!)**

2. **Infinite Do-While Loop**
   ```c
   #include <stdio.h>
   int main() {
       int x = 1;
       do {
           printf("Hello ");
       } while (1);
       return 0;
   }
   ```

3. **Assignment in Condition**
   ```c
   #include <stdio.h>
   int main() {
       int x = 0;
       do {
           printf("Loop ");
       } while (x = 5);
       return 0;
   }
   ```

### **Break and Continue**
4. **Using Break**
   ```c
   #include <stdio.h>
   int main() {
       int i = 0;
       do {
           printf("%d ", i);
           if (i == 2) break;
           i++;
       } while (i < 5);
       return 0;
   }
   ```

5. **Using Continue**
   ```c
   #include <stdio.h>
   int main() {
       int i = 0;
       do {
           i++;
           if (i == 3) continue;
           printf("%d ", i);
       } while (i < 5);
       return 0;
   }
   ```

### **Pointer and Array Manipulation**
6. **Pointer Iteration**
   ```c
   #include <stdio.h>
   int main() {
       char str[] = "Hello";
       char *p = str;
       do {
           printf("%c", *p++);
       } while (*p);
       return 0;
   }
   ```

7. **Array Access**
   ```c
   #include <stdio.h>
   int main() {
       int arr[] = {1, 2, 3, 4, 5};
       int i = 0;
       do {
           printf("%d ", arr[i++]);
       } while (i < 5);
       return 0;
   }
   ```

### **Logical Operators & Conditions**
8. **Logical Operators in Condition**
   ```c
   #include <stdio.h>
   int main() {
       int x = 5, y = 10;
       do {
           printf("%d %d\n", x, y);
       } while (x-- > 0 && ++y < 15);
       return 0;
   }
   ```

9. **Bitwise Operations**
   ```c
   #include <stdio.h>
   int main() {
       int x = 5;
       do {
           printf("%d ", x ^= 1);
       } while (x > 0);
       return 0;
   }
   ```

## **C++ Tricky Do-While Loop Questions**
10. **String Iteration**
    ```cpp
    #include <iostream>
    using namespace std;
    int main() {
        string str = "Code";
        int i = 0;
        do {
            cout << str[i++] << " ";
        } while (i < str.length());
        return 0;
    }
    ```

11. **STL Vector with Do-While**
    ```cpp
    #include <iostream>
    #include <vector>
    using namespace std;
    int main() {
        vector<int> v = {10, 20, 30};
        int i = 0;
        do {
            cout << v[i++] << " ";
        } while (i < v.size());
        return 0;
    }
    ```

12. **Auto Iterator in Do-While**
    ```cpp
    #include <iostream>
    #include <vector>
    using namespace std;
    int main() {
        vector<int> v = {5, 10, 15};
        auto it = v.begin();
        do {
            cout << *it++ << " ";
        } while (it != v.end());
        return 0;
    }
    ```

## **Java Tricky Do-While Loop Questions**
13. **Simple Do-While in Java**
    ```java
    public class Main {
        public static void main(String[] args) {
            int x = 1;
            do {
                System.out.print(x + " ");
            } while (x > 5);
        }
    }
    ```

14. **String Do-While**
    ```java
    public class Main {
        public static void main(String[] args) {
            String str = "Hello";
            int i = 0;
            do {
                System.out.print(str.charAt(i++) + " ");
            } while (i < str.length());
        }
    }
    ```

15. **Nested Do-While**
    ```java
    public class Main {
        public static void main(String[] args) {
            int i = 0, j;
            do {
                j = 0;
                do {
                    System.out.print(i + "" + j + " ");
                    j++;
                } while (j < 3);
                i++;
            } while (i < 3);
        }
    }
    ```

### **Harder Cases**
16. **Negative Counter**
    ```c
    #include <stdio.h>
    int main() {
        int x = -1;
        do {
            printf("%d ", x);
            x--;
        } while (x > -5);
        return 0;
    }
    ```

17. **Odd Number Skip**
    ```java
    public class Main {
        public static void main(String[] args) {
            int x = 1;
            do {
                if (x % 2 == 0) 
                    System.out.print(x + " ");
                x++;
            } while (x <= 10);
        }
    }
    ```

18. **Reverse Printing**
    ```c
    #include <stdio.h>
    int main() {
        int x = 10;
        do {
            printf("%d ", x--);
        } while (x >= 0);
        return 0;
    }
    ```

19. **Boolean Condition**
    ```java
    public class Main {
        public static void main(String[] args) {
            boolean flag = false;
            do {
                System.out.println("Inside Loop");
            } while (flag);
        }
    }
    ```

20. **Floating Point Condition**
    ```java
    public class Main {
        public static void main(String[] args) {
            double x = 1.5;
            do {
                System.out.print(x + " ");
                x -= 0.5;
            } while (x > 0);
        }
    }
    ```
